from openai import OpenAI

client = OpenAI(
    api_key="你的 API Key",
    base_url="https://api.deepseek.com"
)

# 读取本地文件内容
with open("E:\STUDY\CS\启真\类脑\PPT\0612\spike2former.tex", "r", encoding="utf-8") as f:
    file_content = f.read()

# 构建包含文件内容的用户消息
user_message = f"""
请根据以下需求文档，为我总结出核心要点和潜在风险点：
-----
{file_content}
-----
"""

response = client.chat.completions.create(
    model="deepseek-v4-pro",
    messages=[{"role": "user", "content": user_message}]
)

print(response.choices[0].message.content)
