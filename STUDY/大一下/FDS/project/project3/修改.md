#include<stdio.h>
#include<stdlib.h>
#define MAXWEIGHT 200
#define MAXVERTEX 1000
#define INFINITE 100000000

struct AdjVnode {
    int vertex;
    int weight;
    struct AdjVnode *next;
};

typedef struct Vnode {
    struct AdjVnode *firstedge;
} *AdjList[MAXVERTEX];

struct Gnode {
    int nv;
    int ne;
    AdjList G;
};

void BuildGraph(struct Gnode *Graph, int r, int c, int w) {
    struct AdjVnode *s = (struct AdjVnode *)malloc(sizeof(struct AdjVnode));
    s->vertex = c;
    s->weight = w;
    s->next = Graph->G[r]->firstedge;
    Graph->G[r]->firstedge = s;
    
    struct AdjVnode *t = (struct AdjVnode *)malloc(sizeof(struct AdjVnode));
    t->vertex = r;
    t->weight = w;
    t->next = Graph->G[c]->firstedge;
    Graph->G[c]->firstedge = t;
}

struct heapnode {
    int vertex;
    int weight;
};

struct heap {
    struct heapnode *T[MAXVERTEX];
    int inheap[MAXVERTEX];
    int size;
    int cap;
};

// 插入顶点到最小堆
struct heap *InsertHeap(int vertex, struct heap *H, int weight) {
    int i;
    if (H->inheap[vertex] != -1) {
        // 更新已有顶点的权值
        i = H->inheap[vertex];
        if (H->T[i]->weight <= weight) return H;  // 如果新权值不更小，则不更新
        H->T[i]->weight = weight;
    } else {
        // 插入新顶点
        i = H->size++;
        H->T[i] = (struct heapnode *)malloc(sizeof(struct heapnode));
        H->T[i]->vertex = vertex;
        H->T[i]->weight = weight;
        H->inheap[vertex] = i;
    }
    
    // 向上调整堆
    struct heapnode temp = *H->T[i];
    for (; i > 0 && temp.weight < H->T[(i-1)/2]->weight; i = (i-1)/2) {
        H->T[i] = H->T[(i-1)/2];
        H->inheap[H->T[i]->vertex] = i;
    }
    H->T[i] = (struct heapnode *)malloc(sizeof(struct heapnode));
    *H->T[i] = temp;
    H->inheap[temp.vertex] = i;
    
    return H;
}

// 从堆中删除顶点
struct heapnode *DeleteMin(struct heap *H) {
    if (H->size == 0) return NULL;
    
    struct heapnode *min = H->T[0];
    H->inheap[min->vertex] = -1;
    
    if (--H->size > 0) {
        struct heapnode *last = H->T[H->size];
        int i, child;
        
        for (i = 0; i*2+1 < H->size; i = child) {
            child = i*2+1;
            if (child+1 < H->size && H->T[child+1]->weight < H->T[child]->weight)
                child++;
            if (last->weight > H->T[child]->weight) {
                H->T[i] = H->T[child];
                H->inheap[H->T[i]->vertex] = i;
            } else break;
        }
        
        H->T[i] = last;
        H->inheap[last->vertex] = i;
    }
    
    return min;
}

// 判断序列是否为Dijkstra序列
int judge(struct Gnode *Graph, int *s, int nv) {
    int i, j;
    int *dist = (int *)malloc(sizeof(int) * (nv + 1));
    int *collected = (int *)malloc(sizeof(int) * (nv + 1));
    
    // 初始化距离数组和收集数组
    for (i = 1; i <= nv; i++) {
        dist[i] = INFINITE;
        collected[i] = 0;
    }
    
    struct heap *H = (struct heap *)malloc(sizeof(struct heap));
    H->size = 0;
    H->cap = nv;
    for (i = 1; i <= nv; i++) H->inheap[i] = -1;
    
    // 源点的距离为0
    dist[s[1]] = 0;
    H = InsertHeap(s[1], H, 0);
    
    for (i = 1; i <= nv; i++) {
        int u = s[i];
        
        // 如果u已经被收集或者u的距离不是当前最小，返回0
        if (collected[u] || (i > 1 && dist[u] > dist[s[i-1]])) {
            free(dist);
            free(collected);
            while (H->size > 0) {
                free(DeleteMin(H));
            }
            free(H);
            return 0;
        }
        
        collected[u] = 1;
        
        // 遍历u的所有邻接点
        struct AdjVnode *g = Graph->G[u]->firstedge;
        while (g) {
            int v = g->vertex;
            if (!collected[v] && dist[v] > dist[u] + g->weight) {
                dist[v] = dist[u] + g->weight;
                H = InsertHeap(v, H, dist[v]);
            }
            g = g->next;
        }
        
        // 验证当前顶点是否是堆中的最小值
        if (i < nv) {
            struct heapnode *min = DeleteMin(H);
            if (min == NULL || min->vertex != s[i+1]) {
                free(min);
                free(dist);
                free(collected);
                while (H->size > 0) {
                    free(DeleteMin(H));
                }
                free(H);
                return 0;
            }
            free(min);
        }
    }
    
    free(dist);
    free(collected);
    while (H->size > 0) {
        free(DeleteMin(H));
    }
    free(H);
    return 1;
}

int main() {
    int nv, ne, i;
    scanf("%d %d", &nv, &ne);
    
    struct Gnode *Graph = (struct Gnode *)malloc(sizeof(struct Gnode));
    for (i = 0; i < MAXVERTEX; i++) {
        Graph->G[i] = (struct Vnode *)malloc(sizeof(struct Vnode));
        Graph->G[i]->firstedge = NULL;
    }
    Graph->nv = nv;
    Graph->ne = ne;
    
    for (i = 0; i < ne; i++) {
        int r, c, w;
        scanf("%d %d %d", &r, &c, &w);
        BuildGraph(Graph, r, c, w);
    }
    
    int num;
    scanf("%d", &num);
    for (i = 0; i < num; i++) {
        int *s = (int *)malloc(sizeof(int) * (nv + 1));
        for (int j = 1; j <= nv; j++) {
            scanf("%d", &s[j]);
        }
        if (judge(Graph, s, nv) == 1) printf("Yes\n");
        else printf("No\n");
        free(s);
    }
    
    // 释放图的内存
    for (i = 0; i < MAXVERTEX; i++) {
        struct AdjVnode *p = Graph->G[i]->firstedge;
        while (p) {
            struct AdjVnode *temp = p;
            p = p->next;
            free(temp);
        }
        free(Graph->G[i]);
    }
    free(Graph);
    
    return 0;
}
