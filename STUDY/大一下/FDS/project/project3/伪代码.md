    list = new array[nv + 1][2]
    for i from 0 to nv:
        list[i][0] = 200
        list[i][1] = 0
    pm = 200
    for i from 1 to nv:
        min = 200
        if i == 1:
            list[s[i]][0] = 0
        else if list[s[i]][0] > pm:
            for i from 0 to nv:
                free list[i]
            free list
            return 0
        for j from 1 to nv:
            if weight[s[i]][j] != 0 and weight[s[i]][j] + list[s[i]][0] < list[j][0] and list[j][1] == 0:
                list[j][0] = weight[s[i]][j] + list[s[i]][0]
        for k from 1 to nv:
            if list[k][0] < min and list[k][1] == 0 and k != s[i]:
                min = list[k][0]
        pm = min
        list[s[i]][1] = 1
    for i from 0 to nv:
        free list[i]
    free list
    return 1
