input_img = imread('rice.png');

input_img = im2double(input_img);

n = input('缩放倍数 = ');

[p, q] = size(input_img);

p_new = floor(p * n);

q_new = floor(q * n);

output_img = zeros(p_new, q_new);

for k = 1:p_new

for s = 1:q_new

x = k / n;

y = s / n;

a = floor(x);

b = floor(y);

u = x - a;

v = y - b;

a1 = min(a, p);

b1 = min(b, q);

a2 = min(a + 1, p);

b2 = min(b + 1, q);

if a < 1

a1 = 1;

a2 = 1;

end

if b < 1

b1 = 1;

b2 = 1;

end

V11 = input_img(a1, b1);

V12 = input_img(a1, b2);

V21 = input_img(a2, b1);

V22 = input_img(a2, b2);

output_img(k, s) = (1 - u) * (1 - v) * V11 + u * (1 - v) * V21 +(1 - u) * v * V12 +u * v * V22;

end

end

subplot(1, 2, 1);

imshow(input_img);

title('原始图像');

subplot(1, 2, 2);

imshow(output_img);

title('放缩后图像');

imwrite(output_img, 'output.jpg');
