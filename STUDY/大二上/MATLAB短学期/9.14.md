input_img=input('原始图像路径：','s');

I=imread(input_img);

figure, imshow(I), title('原始图像');

gray=rgb2gray(I);

BW=edge(gray,'canny');

[H,theta,rho]=hough(BW);

P=houghpeaks(H,20,'threshold',ceil(0.3*max(H(:))));

lines=houghlines(BW,theta,rho,P,'FillGap',20,'MinLength',30);

len=zeros(length(lines),1);

for k=1:length(lines)

xy=[lines(k).point1; lines(k).point2];

len(k)=norm(lines(k).point1 - lines(k).point2);

end

[~,idx]=maxk(len,3);

figure, imshow(I), hold on, title('最长的3条直线');

for k=idx'

xy=[lines(k).point1; lines(k).point2];

plot(xy(:,1),xy(:,2),'LineWidth',3,'Color','red');

plot(xy(1,1),xy(1,2),'x','LineWidth',2,'Color','yellow');

plot(xy(2,1),xy(2,2),'x','LineWidth',2,'Color','green');

end

hold off;

BW2=imbinarize(gray);

BW2=imfill(BW2,'holes');

BW2=bwareaopen(BW2,50);

[B,L]=bwboundaries(BW2,'noholes');

stats=regionprops(L,'Area');

allArea=[stats.Area];

[~,idx2]=maxk(allArea,3);

figure, imshow(I), hold on, title('面积最大的3个区域');

colors=lines(3);

for k=1:3

boundary=B{idx2(k)};

plot(boundary(:,2),boundary(:,1),'Color',colors(k,:),'LineWidth',3);

end

hold off;
