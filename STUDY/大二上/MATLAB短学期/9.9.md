a=imread('peppers.png');
subplot(3,2,1);imshow(a);
xlabel('(a)原始图像');
b1=a+50;
subplot(3,2,2);imshow(b1);
xlabel('(b)灰度值增大');
b2=a* 1.2;
subplot(3,2,3);imshow(b2);
xlabel('(c)对比度增大');
b3=a* 0.65;
subplot(3,2,4);imshow(b3);
xlabel('(d)对比度减小');
b4=-double(a)+255;
subplot(3,2,5);imshow(uint8(b4));
xlabel('(e)图像求补运算');



a=imread('peppers.png');

xlabel('(a)原始图像');
x=1:255;
y=x+x.* (255-x)/255;
subplot(1,3,2);plot(x,y);
xlabel('(b)函数的曲线图');
b1=double(a)+0.006* double(a).* (255-double(a));
subplot(1,3,3);imshow(uint8(b1));
xlabel('(c)非线性处理结果');
